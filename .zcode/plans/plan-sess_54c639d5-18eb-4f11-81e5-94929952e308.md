## 第一部分：后端 RBAC 接口级强制（TDD 核心）

### Step 1 — 优化权限查询（消除 N+1）

**文件**：`backend/app/services/auth_service.py`

将 `get_user_permission_codes` 从"逐 role 逐 permission 循环查询"改为**单条 4 表 join**：

```python
async def get_user_permission_codes(db: AsyncSession, user_id: int) -> list[str]:
    result = await db.execute(
        select(Permission.code)
        .join(RolePermission, RolePermission.permission_id == Permission.id)
        .join(Role, Role.id == RolePermission.role_id)
        .join(UserRole, UserRole.role_id == Role.id)
        .where(UserRole.user_id == user_id)
    )
    return list(result.scalars().all())
```

同时新增 `get_user_role_codes`（用于 super 豁免判断）：
```python
async def get_user_role_codes(db: AsyncSession, user_id: int) -> list[str]:
    result = await db.execute(
        select(Role.code).join(UserRole, UserRole.role_id == Role.id).where(UserRole.user_id == user_id)
    )
    return list(result.scalars().all())
```

### Step 2 — 新增 `require_permission` 依赖（dependencies.py）

**文件**：`backend/app/core/dependencies.py`

```python
def require_permission(*permission_codes: str):
    """菜单级权限校验依赖工厂。
    
    用法：def handler(db: DbSession, user: CurrentUser, _=Depends(require_permission("devices"))):
    或在 APIRouter(dependencies=[require_permission("devices")]) 批量挂载。
    
    super 角色豁免（拥有所有权限）。
    """
    async def _check_permission(
        user: CurrentUser,           # 复用现有鉴权链（先验 token→403）
        db: AsyncSession,
    ):
        role_codes = await get_user_role_codes(db, user.id)
        if "super" in role_codes:
            return  # 超管豁免
        user_perms = set(await get_user_permission_codes(db, user.id))
        if not user_perms.intersection(permission_codes):
            raise BusinessException(code=403, message=f"权限不足，需要: {', '.join(permission_codes)}")
    return _check_permission
```

```typescript
async function fetchStatusHistory(id, _params?) {
  const res = await getMeterDetail(id);           // ❌ 调错端点

// 修正后
import { getMeterStatusHistory } from '#/api/modules/meter';  // 已在 import 列表
async function fetchStatusHistory(id, _params?) {
  statusHistoryLoading.value = true;
  try {
    const res = await getMeterStatusHistory(id);  // ✅ 专用端点
    statusHistory.value = res.items || res || [];  // ✅ 适配返回格式
  } finally {
    statusHistoryLoading.value = false;
  }
}
```
